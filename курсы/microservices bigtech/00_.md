- Makefile
	содержит команды с линтером install + запуск
	метод ==`run`== + ==`./...`==  проходит по всем папкам и подпапкам...
	==`--config`== - указываем конфиг, иначе будет использовать дефолтный
```
LOCAL_BIN:=$(CURDIR)/bin

install-golangci-lint:
	GOBIN=$(LOCAL_BIN) go install github.com/golangci/golangci-lint/cmd/golangci-lint@v1.61.0

lint:
	$(LOCAL_BIN)/golangci-lint run ./... --config .golangci.pipeline.yaml
```

- ==`.golangci.pipeline.yaml`==
	сам конфиг линта
	
- ==`.github/workflows`==
	для github actions
	ci процессы (билдим тестим линтим)
	- ==`go.yaml`==
```yaml
name: Go

on: # когда запускать экшины
  push:  # если ктото пушит в ветку main, master
    branches: [ main, master ] 
  pull_request: # или будет пул реквест
    branches: [ main, master ]
	
jobs: # шаги которые будут запускаться
  build-and-test:  # первая джоба
    runs-on: ubuntu-latest  # за основу берем последнюю версию убунту
    steps: # степы
        - uses: actions/checkout@v4 # использовуем уже существующий экшин. чекаут на нужную ветку
		
        - name: Set up Go # новый экшин
          uses: actions/setup-go@v5 # подготовит в убунте golang c нужной версией и зависимостями..
          with:
            go-version: '1.23'
            cache-dependency-path: go.sum
			
        - name: Build # шаг с билдом main
          run: go build -o ./bin/ -v ./...
		
        - name: Test # запуск тестов по всему проектов если нет то в холостую пройдется
          run: go test -v ./...
		
	linter: # джоба 2
    name: lint # 
    runs-on: ubuntu-latest # берем последнюю версию убунту
    steps:
      - uses: actions/checkout@v4 # переключаемся на нужную ветку
      - uses: actions/setup-go@v5 # настройка под го
        with:
          go-version: '1.23'
          cache: false
      - name: golangci-lint
        uses: golangci/golangci-lint-action@v6 # будем использовать экшн от голинт
        with:
          # Require: The version of golangci-lint to use.
          # When `install-mode` is `binary` (default) the value can be v1.2 or v1.2.3 or `latest` to use the latest version.
          # When `install-mode` is `goinstall` the value can be v1.2.3, `latest`, or the hash of a commit.
          version: v1.61 #версия линта

          # Optional: working directory, useful for monorepos
          # working-directory: somedir

          # Optional: golangci-lint command line arguments.
          #
          # Note: By default, the `.golangci.yml` file should be at the root of the repository.
          # The location of the configuration file can be changed by using `--config=`
          args: --timeout=30m --config=./.golangci.pipeline.yaml # в аргументе указали таймаут и откуда брать конфиг для линта
```
